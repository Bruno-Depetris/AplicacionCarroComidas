CREATE TABLE Caja (
    id_caja INTEGER PRIMARY KEY AUTOINCREMENT,
    fecha_apertura DATETIME NOT NULL,
    hora_cierre DATETIME,
    cambio_inicial REAL NOT NULL,
    cambio_final REAL,
    estado TEXT CHECK(estado IN ('ABIERTA', 'CERRADA')) NOT NULL
);
CREATE TABLE Producto (
    id_producto INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    precio REAL NOT NULL
);
CREATE TABLE Venta (
    id_venta INTEGER PRIMARY KEY AUTOINCREMENT,
    fecha DATETIME NOT NULL,
    total REAL NOT NULL,
    estado_cobro TEXT CHECK(estado_cobro IN ('PENDIENTE', 'COBRADO')) NOT NULL,
    metodo_pago TEXT CHECK(metodo_pago IN ('EFECTIVO', 'TRANSFERENCIA', 'DEBITO', '')) NOT NULL DEFAULT '',
    id_caja INTEGER NOT NULL,
    FOREIGN KEY (id_caja) REFERENCES Caja(id_caja)
);
CREATE TABLE DetalleVenta (
    id_detalle INTEGER PRIMARY KEY AUTOINCREMENT,
    id_venta INTEGER NOT NULL,
    Detalle INTEGER NOT NULL, (todos los productos que compro)
    cantidad INTEGER NOT NULL,
    subtotal REAL NOT NULL,
    FOREIGN KEY (id_venta) REFERENCES Venta(id_venta),
    FOREIGN KEY (id_producto) REFERENCES Producto(id_producto)
);
CREATE TABLE Movimiento (
    id_movimiento INTEGER PRIMARY KEY AUTOINCREMENT,
    fecha DATETIME NOT NULL,
    tipo TEXT CHECK(tipo IN ('INGRESO', 'RETIRO')) NOT NULL,
    motivo TEXT,
    monto REAL NOT NULL,
    metodo TEXT CHECK(metodo IN ('EFECTIVO', 'TRANSFERENCIA', 'DEBITO')) NOT NULL,
    id_caja INTEGER NOT NULL,
    FOREIGN KEY (id_caja) REFERENCES Caja(id_caja)
);
CREATE VIEW HistorialVentas AS
SELECT 
    V.id_venta,
    V.fecha,
    GROUP_CONCAT(P.nombre || ' x' || DV.cantidad, ', ') AS productos,
    V.total,
    V.estado_cobro,
    V.metodo_pago
FROM Venta V
JOIN DetalleVenta DV ON V.id_venta = DV.id_venta
JOIN Producto P ON DV.id_producto = P.id_producto
GROUP BY V.id_venta;
